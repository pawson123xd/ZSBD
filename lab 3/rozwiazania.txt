CREATE VIEW zadanie3 AS
 select    
    e.department_id,
    d.department_name,
    COUNT(d.department_name) AS liczba_pracownikow,
    AVG(e.salary) AS srednia_pensja,
    MAX(e.salary) AS najwyzsza_pensja
    from employees e
    inner join departments  d on d.department_id=e.department_id
    GROUP  by      e.department_id,d.department_name
    having COUNT(d.department_name) >= 4

CREATE VIEW zadanie9 AS
SELECT 
    e.employee_id,e.first_name,e.last_name,d.department_name FROM employees e
JOIN departments d ON e.department_id = d.department_id
WHERE e.employee_id IN (
    SELECT manager_id
    FROM employees
    WHERE manager_id IS NOT NULL
);

CREATE VIEW zadanie10 AS

FROM employees e
