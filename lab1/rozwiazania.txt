where e.department_id = 20 or e.department_id=50 and e.salary between 2000 and 7000 ORDER by e.last_name


select e.hire_date,e.last_name, &kolumna from employees e
where e.manager_id is not null and e.hire_date >=DATE '2005-01-01' and e.hire_date < DATE '2006-01-01' 
order by  &kolumna;

select  e.first_name || ' ' || e.last_name AS imie_nazwisko , e.salary,e.phone_number  from employees e
where e.last_name like '__e' and e.first_name like 'Da%'
ORDER BY 1 DESC, 2 ASC;

select e.first_name || '' || e.last_name as imie_nazwisko,   ROUND(MONTHS_BETWEEN(SYSDATE, hire_date)) AS liczba_miesiecy, 
CASE 
when
 MONTHS_BETWEEN(SYSDATE, hire_date) < 150 
 THEN  e.salary*0.10
when
 MONTHS_BETWEEN(SYSDATE, hire_date) BETWEEN 150 AND 200
 THEN  e.salary*0.20
 when
 MONTHS_BETWEEN(SYSDATE, hire_date) > 200
 THEN  e.salary*0.30
 END AS wysokosc_dodatku
from employees e
ORDER BY liczba_miesiecy;


SELECT 
    SUM(e.salary) AS suma_zarobkow, 
    ROUND(AVG(e.salary)) AS srednia_zarobkow 
FROM 
    employees e 
GROUP BY 
    e.department_id 
HAVING 
    MIN(e.salary) > 5000;

SELECT
    r.region_name,
    COUNT( c.country_id),
    COUNT( e.employee_id)
FROM regions r
LEFT JOIN countries c   ON r.region_id = c.region_id
LEFT JOIN locations l   ON c.country_id = l.country_id
LEFT JOIN departments d ON l.location_id = d.location_id
LEFT JOIN employees e   ON d.department_id = e.department_id
GROUP BY r.region_name;
